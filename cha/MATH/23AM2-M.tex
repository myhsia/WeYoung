\section{计算题}
\begin{exercise}
    求过点$A=\ab(1,2,3),\ B\ab(3,4,5),\ C\ab(2,2,3)$三点的平面方程.
\end{exercise}

\begin{exercise}
    设$z=xy+xF\ab(\frac{y}{x})$，而$F\ab(x)$为可导函数.求$\frac{\partial z}{\partial x},\ \frac{\partial z}{\partial y}$.
\end{exercise}

\begin{exercise}
    求函数$z=3x^2+y^2+2xy-10x-2y+7$的极值.
\end{exercise}

\begin{exercise}
    计算$I=\iint\limits_{D}{x\d x\d y}$，其中$D$为由$y=x^2$及$y=x$所围成的区域.
\end{exercise}

\begin{exercise}
    求直线$L:\ 
    \left\{
        \begin{aligned}
            &x+y+2z-1=0\\
            &3x-y+z-1=0
        \end{aligned}
    \right.
    $在平面$\Pi:\ x+2y-z=0$上的投影直线.
\end{exercise}

\begin{exercise}
    设$u=f\ab(x,y,z),\ \varphi\ab(x^2,\ee^y,z)=0,\ y=\sin{x}$.其中函数$f,\ \varphi$具有一阶连续偏导数，且$\frac{\partial\varphi}{\partial z}\neq 0$.求$\frac{\d u}{\d x}$.
\end{exercise}

\begin{exercise}
    计算三重积分$\iiint\limits_{\Omega}{\sqrt{y^2+z^2}}\d x\d y\d z$.其中$\Omega$由$xOy$平面上的曲线$x=hy$（$h>0$）绕$x$轴旋转而成的曲面与平面$x=h$所围成的闭区域.
\end{exercise}

\begin{exercise}
    求双曲抛物面$z=xy$被柱面$x^2+y^2=2$所截出的那部分面积.
\end{exercise}

\begin{exercise}
    已知函数$f\ab(x,y)=
    \left\{
        \begin{aligned}
            &\ab(x+y)\sin{\frac{1}{x+y}}&&x^2+y^2\neq 0\\
            &0&&x^2+y^2=0
        \end{aligned}
    \right.
    $，问
    \vspace{1em}
    \begin{enumerate}
        \item 函数$f$在点$\ab(0,0)$处是否连续
        \item 函数$f$在点$\ab(0,0)$处的偏导数是否存在，如果存在，则求出$f_x\ab(0,0)$和$f_y\ab(0,0)$.
    \end{enumerate}
\end{exercise}

\begin{exercise}
    设函数$f\ab(x)$在$\comm{a}{b}$上连续，证明不等式
    $$\ab[\int_a^b{f\ab(x)\d x}]^2\le\ab(b-a)\int_a^b{f^2\ab(x)\d x}$$
\end{exercise}
